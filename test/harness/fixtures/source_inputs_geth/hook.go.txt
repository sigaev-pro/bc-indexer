package core

import (
	"github.com/ethereum/go-ethereum/common"
	"github.com/ethereum/go-ethereum/core/types"
)

var extractorGetHash func(*types.Header, func(uint64) common.Hash) func(uint64) common.Hash

// SetExtractorGetHash installs f. The serial state processor calls f once per
// block, after it creates the block EVM and before the first transaction, and
// uses the returned function for every BLOCKHASH lookup of that block. The
// state prefetcher and the parallel processor never call it.
func SetExtractorGetHash(f func(*types.Header, func(uint64) common.Hash) func(uint64) common.Hash) {
	extractorGetHash = f
}
